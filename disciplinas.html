let simuladoQuestoes = []; let simuladoIndice = 0; let simuladoAcertos = 0; function abrirSimulado() { simuladoQuestoes = criarQuestoes(); if (simuladoQuestoes.length > 5) { simuladoQuestoes = simuladoQuestoes.slice(0, 5); } simuladoIndice = 0; simuladoAcertos = 0; document.getElementById("areaEstudo").innerHTML = `

🎯 Simulado

${materiaEscolhida} — ${serieEscolhida}º ano

`; mostrarQuestaoSimulado(); document.getElementById("areaEstudo").scrollIntoView({ behavior: "smooth" }); } function mostrarQuestaoSimulado() { const container = document.getElementById("simuladoContainer"); if (!container) return; if (simuladoIndice >= simuladoQuestoes.length) { finalizarSimulado(); return; } const q = simuladoQuestoes[simuladoIndice]; container.innerHTML = `

Questão ${simuladoIndice + 1} de ${simuladoQuestoes.length}

${q.pergunta}

${q.alternativas.map((alt, index) => ` `).join("")}
`; } function responderSimulado(resposta) { const q = simuladoQuestoes[simuladoIndice]; if (resposta === q.correta) { simuladoAcertos++; } simuladoIndice++; mostrarQuestaoSimulado(); } function finalizarSimulado() { const total = simuladoQuestoes.length; const erros = total - simuladoAcertos; const porcentagem = Math.round((simuladoAcertos / total) * 100); if (typeof registrarResultadoAtual === "function") { registrarResultadoAtual( simuladoAcertos, erros, total ); } document.getElementById("simuladoContainer").innerHTML = `

🏆 Simulado concluído!

Acertos: ${simuladoAcertos}

Erros: ${erros}

Aproveitamento: ${porcentagem}%

Ver resultados
`; }
